set(TWOINE_UNICORN_SOURCE_DIR "" CACHE PATH "Local Unicorn source tree for an embedded build")
if(TWOINE_UNICORN_SOURCE_DIR)
    set(UNICORN_ARCH "x86" CACHE STRING "Unicorn guest architecture" FORCE)
    set(UNICORN_BUILD_TESTS OFF CACHE BOOL "Unicorn tests" FORCE)
    set(UNICORN_INSTALL OFF CACHE BOOL "Unicorn install" FORCE)
    set(UNICORN_LEGACY_STATIC_ARCHIVE OFF CACHE BOOL "Unicorn archive bundling" FORCE)
    set(BUILD_SHARED_LIBS OFF)
    add_subdirectory("${TWOINE_UNICORN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/unicorn" EXCLUDE_FROM_ALL)
    set(TWOINE_UNICORN_TARGET unicorn)
else()
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(UNICORN QUIET IMPORTED_TARGET unicorn>=2.1.4)
endif()
if(TARGET PkgConfig::UNICORN)
    set(TWOINE_UNICORN_TARGET PkgConfig::UNICORN)
else()
    find_path(UNICORN_INCLUDE_DIR unicorn/unicorn.h)
    find_library(UNICORN_LIBRARY NAMES unicorn unicorn_static)
    if(NOT UNICORN_INCLUDE_DIR OR NOT UNICORN_LIBRARY)
        message(FATAL_ERROR "Unicorn >=2.1.4 is required for the portable CPU runtime")
    endif()
    add_library(twoine_unicorn UNKNOWN IMPORTED)
    set_target_properties(twoine_unicorn PROPERTIES IMPORTED_LOCATION "${UNICORN_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${UNICORN_INCLUDE_DIR}")
    set(TWOINE_UNICORN_TARGET twoine_unicorn)
endif()

endif()

add_library(twoine_runtime_objects OBJECT cpu.c cpu_unicorn.c ${TWOINE_HOST_SOURCES}
    execution.c execution_memory.c execution_unicorn.c ../guest/arguments.c)
set_target_properties(twoine_runtime_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(twoine_runtime_objects PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(twoine_runtime_objects PUBLIC c_std_99)
target_compile_definitions(twoine_runtime_objects PRIVATE TWOINE_WITH_UNICORN=1)
find_package(Threads REQUIRED)
target_link_libraries(twoine_runtime_objects PUBLIC Threads::Threads ${TWOINE_UNICORN_TARGET} ${CMAKE_DL_LIBS})
if(MSVC)
    target_compile_options(twoine_runtime_objects PRIVATE /W4)
else()
    target_compile_options(twoine_runtime_objects PRIVATE -Wall -Wextra -Werror)
endif()
add_library(twoine_runtime STATIC $<TARGET_OBJECTS:twoine_runtime_objects>)
target_include_directories(twoine_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(twoine_runtime PUBLIC c_std_99)
target_link_libraries(twoine_runtime PRIVATE Threads::Threads ${TWOINE_UNICORN_TARGET} ${CMAKE_DL_LIBS})
if(BUILD_TESTING)
    add_executable(twoine_host_sync_tests ../tests/runtime/test_host_sync.c)
    target_link_libraries(twoine_host_sync_tests PRIVATE twoine_runtime Threads::Threads)
    add_test(NAME host_sync COMMAND twoine_host_sync_tests)
    set_tests_properties(host_sync PROPERTIES TIMEOUT 20)
    add_executable(twoine_handle_tests ../tests/runtime/test_handles.c ../guest/handles.c host_sync.c host.c)
    target_include_directories(twoine_handle_tests PRIVATE ../guest)
    target_link_libraries(twoine_handle_tests PRIVATE Threads::Threads)
    add_test(NAME guest_handles COMMAND twoine_handle_tests)
    set_tests_properties(guest_handles PROPERTIES TIMEOUT 20)
    add_executable(twoine_execution_tests ../tests/runtime/test_execution.c)
    target_link_libraries(twoine_execution_tests PRIVATE twoine_runtime)
    add_test(NAME runtime_execution COMMAND twoine_execution_tests unicorn)
    set_tests_properties(runtime_execution PROPERTIES TIMEOUT 20)
    add_executable(twoine_runtime_tests ../tests/runtime/test_cpu.c)
    target_link_libraries(twoine_runtime_tests PRIVATE twoine_runtime)
    add_test(NAME runtime_cpu COMMAND twoine_runtime_tests)
    set_tests_properties(runtime_cpu PROPERTIES TIMEOUT 20)
endif()
