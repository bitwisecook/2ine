foreach(module doscalls msg nls quecalls viocalls kbdcalls sesmgr som pmwin pmshapi pmgpi tcpip32)
    if(module STREQUAL "doscalls")
        set(sources doscalls/module.c doscalls/files.c doscalls/fileinfo.c doscalls/directories.c doscalls/paths.c doscalls/resources.c doscalls/exceptions.c doscalls/memory.c doscalls/threads.c doscalls/process.c doscalls/semaphores.c doscalls/modules.c doscalls/system.c doscalls/audio.c)
    elseif(module STREQUAL "pmwin")
        set(sources pmwin/module.c pmwin/context.c pmwin/windows.c pmwin/messages.c pmwin/paint.c)
    else()
        set(sources "${module}.c")
    endif()
    add_library(${module} SHARED ${sources})
    twoine_application_target(${module})
    target_link_libraries(${module} PRIVATE 2ine)
    set_target_properties(${module} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
    if(APPLE)
        set_target_properties(${module} PROPERTIES SUFFIX ".so")
    endif()
endforeach()
if(TWOINE_NATIVE_LOADER)
    twoine_hal_add_native_abi(doscalls doscalls)
    twoine_hal_add_native_abi(msg msg)
endif()
twoine_add_terminal(viocalls)
