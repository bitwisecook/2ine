            if(NOT EXISTS "${TWOINE_${_twoine_gui_name}_EXE}")

                --work "${PROJECT_BINARY_DIR}/gui-results/${profile}")
            set_tests_properties(gui_${profile} PROPERTIES TIMEOUT 30)
